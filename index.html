/**

 * Paste this code into: Extensions > Apps Script in your Google Spreadsheet

// Automatically creates all 5 tabs when the spreadsheet is opened
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu("IT Operations Hub")
    .addItem("Initialize / Verify All 5 Tabs", "menuInitializeSheets")
    .addToUi();
  initializeAllSheets(SpreadsheetApp.getActiveSpreadsheet());
}

function menuInitializeSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  initializeAllSheets(ss);
  SpreadsheetApp.getUi().alert("Success! All 5 tabs (Tickets, Assets, Licenses, Projects, Daily_Routine) have been created and verified.");
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Always guarantee all 5 tabs exist before performing any action
    initializeAllSheets(ss);

    var data = JSON.parse(e.postData.contents);
    var action = data.action;

    // ==========================================
    // 1. HELPDESK TICKETS
    // ==========================================
    if (action === "ADD_TICKET") {
      var sheet = ss.getSheetByName("Tickets");
      var t = data.data;
      var now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
      sheet.appendRow([
        t.id, t.createdDate, t.name, t.email, t.dept,
        t.category, t.priority, t.assigned, t.desc, t.status, t.remarks || "", now
      ]);
      return createJsonResponse({ result: "success", id: t.id });
    }

    if (action === "UPDATE_TICKET_STATUS" || action === "RESOLVE_TICKET") {
      var sheet = ss.getSheetByName("Tickets");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        var targetId = data.id;
        var newStatus = data.status || (action === "RESOLVE_TICKET" ? "Resolved" : "Pending");
        var remarks = data.remarks || "";
        var now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");

        for (var i = 1; i < rows.length; i++) {
          if (rows[i][0] == targetId) {
            sheet.getRange(i + 1, 10).setValue(newStatus); // Status column
            if (remarks) sheet.getRange(i + 1, 11).setValue(remarks); // Resolution Remarks
            sheet.getRange(i + 1, 12).setValue(now); // Last Updated
            
            var userEmail = rows[i][3];
            var userName = rows[i][2];
            if (newStatus === "Resolved" && userEmail) {
              try {
                GmailApp.sendEmail(
                  userEmail,
                  "Re: [IT-Hub] Ticket #" + targetId + " - Resolved",
                  "Hi " + userName + ",\n\nYour IT support ticket #" + targetId + " has been resolved.\n\nResolution: " + remarks + "\n\nRegards,\nIT Operations Team"
                );
              } catch(e) {}
            }
            break;
          }
        }
      }
      return createJsonResponse({ result: "success" });
    }

    // ==========================================
    // 2. ASSET INVENTORY
    // ==========================================
    if (action === "ADD_ASSET_BULK") {
      var sheet = ss.getSheetByName("Assets");
      var list = data.data || [];
      var lastRow = sheet.getLastRow();
      list.forEach(function(a, idx) {
        sheet.appendRow([
          lastRow + idx,
          a.tag,
          a.serial,
          a.classification,
          a.assetType,
          a.model,
          a.configuration,
          a.warrantyStart,
          a.warrantyEnd,
          a.value,
          a.empId,
          a.empName,
          a.manager,
          a.dept,
          a.assignedDate,
          a.vendor,
          a.status,
          a.location,
          a.licenseKey,
          a.remarks
        ]);
      });
      return createJsonResponse({ result: "success", count: list.length });
    }

    if (action === "UPDATE_ASSET_ASSIGNMENT") {
      var sheet = ss.getSheetByName("Assets");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        var a = data.data;
        for (var i = 1; i < rows.length; i++) {
          if (rows[i][1] == a.tag) { // ASSET ID CODE column
            sheet.getRange(i + 1, 11).setValue(a.empId);
            sheet.getRange(i + 1, 12).setValue(a.empName);
            sheet.getRange(i + 1, 13).setValue(a.manager);
            sheet.getRange(i + 1, 14).setValue(a.dept);
            sheet.getRange(i + 1, 15).setValue(a.assignedDate);
            sheet.getRange(i + 1, 17).setValue(a.status);
            sheet.getRange(i + 1, 18).setValue(a.location);
            sheet.getRange(i + 1, 20).setValue(a.remarks);
            break;
          }
        }
      }
      return createJsonResponse({ result: "success" });
    }

    // ==========================================
    // 3. SOFTWARE & LICENSES
    // ==========================================
    if (action === "ADD_LICENSE" || action === "UPDATE_LICENSE") {
      var sheet = ss.getSheetByName("Licenses");
      var l = data.data;
      var existingRow = -1;
      var rows = sheet.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] == l.id) { existingRow = i + 1; break; }
      }
      
      var rowData = [
        l.id, l.softwareName, l.category, l.acquisitionType, l.totalSeats,
        l.licenseKey, l.portalUrl, l.currency, l.cost, l.purchaseDate,
        l.isLifetime ? "YES" : "NO", l.expiryDate || "N/A", l.assignedScope,
        l.assignedDept, l.assignedUser, l.status
      ];

      if (existingRow > 0) {
        sheet.getRange(existingRow, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
      return createJsonResponse({ result: "success" });
    }

    // ==========================================
    // 4. PROJECTS TRACKER
    // ==========================================
    if (action === "ADD_PROJECT" || action === "UPDATE_PROJECT") {
      var sheet = ss.getSheetByName("Projects");
      var p = data.data;
      var now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
      var existingRow = -1;
      var rows = sheet.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] == p.id) { existingRow = i + 1; break; }
      }

      var rowData = [p.id, p.title, p.lead, p.dueDate, p.status, p.progress, p.remarks, now];
      if (existingRow > 0) {
        sheet.getRange(existingRow, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
      return createJsonResponse({ result: "success" });
    }

    // ==========================================
    // 5. DAILY ROUTINE SHIFT LOGS
    // ==========================================
    if (action === "ADD_ROUTINE") {
      var sheet = ss.getSheetByName("Daily_Routine");
      var r = data.data;
      var now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
      sheet.appendRow([
        r.id, r.date, r.techName, r.shift, r.completed, r.pending || "-", r.blockers || "-", now
      ]);
      return createJsonResponse({ result: "success" });
    }

    return createJsonResponse({ result: "ok", message: "No matching action" });

  } catch (err) {
    return createJsonResponse({ result: "error", error: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  initializeAllSheets(ss);
  return createJsonResponse({ status: "active", message: "IT Operations Webhook is running. All 5 tabs initialized." });
}

// Guarantee creation and header formatting for all 5 tabs
function initializeAllSheets(ss) {
  getOrCreateSheet(ss, "Tickets", [
    "Ticket ID", "Created Date", "Requester Name", "Email", "Department",
    "Category", "Priority", "Assign ID", "Description", "Status", "Resolution Remarks", "Last Updated"
  ]);

  getOrCreateSheet(ss, "Assets", [
    "SL NO", "ASSET ID CODE", "Serial Number", "Classification", "Asset Type",
    "Model Name", "Configuration", "Warranty Start", "Warranty End", "Value",
    "Employee ID", "Employee Name", "Manager", "Department", "Assigned Date",
    "VENDOR", "Status", "Location", "License key", "Remarks"
  ]);

  getOrCreateSheet(ss, "Licenses", [
    "License ID", "Software Name", "Category", "Acquisition Type", "Total Seats",
    "License Key", "Vendor Portal URL", "Currency", "Cost", "Purchase Date",
    "Is Lifetime", "Expiry Date", "Assigned Scope", "Assigned Dept", "Assigned Users", "Status"
  ]);

  getOrCreateSheet(ss, "Projects", [
    "Project ID", "Title", "Lead", "Target Due Date", "Status", "Progress (%)", "Status Remarks", "Last Updated"
  ]);

  getOrCreateSheet(ss, "Daily_Routine", [
    "Log ID", "Date", "Technician Name", "Shift", "Completed Routine Work", "Pending Follow-ups", "Blockers / Outages", "Logged At"
  ]);

  // Remove default "Sheet1" if empty and other sheets exist
  var defaultSheet = ss.getSheetByName("Sheet1");
  if (defaultSheet && ss.getSheets().length > 1 && defaultSheet.getLastRow() === 0) {
    try { ss.deleteSheet(defaultSheet); } catch(e) {}
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
    var headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold");
  }
  return sheet;
}

function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
